Troubleshooting Guide
Production Incident #1: Database Connection Pool Exhaustion

Symptoms:

  • API response times spike to 30+ seconds
  • Error logs show "Too many connections" or "Connection timeout"
  • Database CPU usage at 100%
  • Multiple endpoints returning 503 errors

Root Cause:

Connection pool size too small for traffic volume, or connections not being released properly due to unhandled exceptions.

Resolution:

// Increase connection pool size in database config
const mongoose = require('mongoose');

mongoose.connect(process.env.MONGODB_URI, {
  maxPoolSize: 50,  // Increase from default 5
  minPoolSize: 10,
  serverSelectionTimeoutMS: 5000,
  socketTimeoutMS: 45000,
});

// Ensure connections are released in error handlers
app.use(async (req, res, next) => {
  try {
    await next();
  } catch (error) {
    // Close any open database connections
    if (mongoose.connection.readyState === 1) {
      // Connection is open, will be released automatically
    }
    next(error);
  }
});
Production Incident #2: Memory Leak in Agent Orchestration

Symptoms:

  • Node.js process memory grows continuously over 24-48 hours
  • Event loop lag increases, causing slow responses
  • OOM killer terminates the process
  • Heap snapshot shows thousands of unresolved promises

Root Cause:

Agent spawn operations not cleaning up event listeners and child process references after completion.

Resolution:

// Proper cleanup in agent orchestration
class AgentOrchestrator {
  async spawnAgent(agentConfig) {
    const child = fork(agentConfig.script, agentConfig.args);

    // Store reference for cleanup
    this.activeAgents.set(child.pid, child);

    return new Promise((resolve, reject) => {
      child.on('message', (result) => {
        this.cleanupAgent(child.pid);
        resolve(result);
      });

      child.on('error', (err) => {
        this.cleanupAgent(child.pid);
        reject(err);
      });

      child.on('exit', (code) => {
        this.cleanupAgent(child.pid);
        if (code !== 0) reject(new Error(`Agent exited with code ${code}`));
      });
    });
  }

  cleanupAgent(pid) {
    const agent = this.activeAgents.get(pid);
    if (agent) {
      agent.removeAllListeners();
      agent.kill();
      this.activeAgents.delete(pid);
    }
  }
}
Production Incident #3: Workflow Stuck in Infinite Loop

Symptoms:

  • Workflow execution never completes
  • CPU usage spikes on workflow runner
  • Logs show same agent being spawned repeatedly
  • Database shows thousands of workflow step records

Root Cause:

Circular dependency in workflow graph where agent A triggers agent B which triggers agent A again.

Resolution:

// Add cycle detection to workflow engine
class WorkflowEngine {
  constructor() {
    this.executionStack = new Set();
    this.maxDepth = 10;
  }

  async executeStep(step, depth = 0) {
    // Prevent infinite loops
    if (depth > this.maxDepth) {
      throw new Error(`Workflow exceeded max depth of ${this.maxDepth}`);
    }

    const stepKey = `${step.agent}-${step.inputHash}`;
    if (this.executionStack.has(stepKey)) {
      throw new Error(`Circular dependency detected: ${stepKey}`);
    }

    this.executionStack.add(stepKey);

    try {
      const result = await this.runAgent(step.agent, step.input);

      // Execute next steps
      for (const nextStep of step.nextSteps || []) {
        await this.executeStep(nextStep, depth + 1);
      }

      return result;
    } finally {
      this.executionStack.delete(stepKey);
    }
  }
}
Production Incident #4: API Rate Limiting Not Working

Symptoms:

  • Single client making thousands of requests per minute
  • Server resources exhausted, affecting all users
  • Rate limit headers not present in responses
  • Logs show rate limiter middleware not being invoked

Root Cause:

Rate limiter middleware applied after route handlers instead of before, or Redis connection failing silently.

Resolution:

// Apply rate limiter BEFORE routes
const rateLimit = require('express-rate-limit');
const RedisStore = require('rate-limit-redis');
const Redis = require('ioredis');

const redisClient = new Redis(process.env.REDIS_URL);

const apiLimiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 minutes
  max: 100, // limit each IP to 100 requests per windowMs
  store: new RedisStore({
    sendCommand: (...args) => redisClient.call(...args),
  }),
  standardHeaders: true, // Return rate limit info in headers
  legacyHeaders: false,
  handler: (req, res) => {
    res.status(429).json({
      error: 'Too many requests, please try again later',
      retryAfter: res.getHeader('Retry-After')
    });
  }
});

// Apply to all API routes FIRST
app.use('/api/', apiLimiter);
app.use('/api/', apiRoutes);
Production Incident #5: Metrics Collection Failure

Symptoms:

  • Dashboard shows no data or stale metrics
  • metrics.json file not being updated
  • Hook executions completing but not recorded
  • Disk space warnings on metrics volume

Root Cause:

Metrics writer process crashing due to file permissions or disk full, or concurrent writes corrupting JSON file.

Resolution:

// Atomic metrics updates with file locking
const fs = require('fs').promises;
const path = require('path');

class MetricsWriter {
  constructor(metricsPath) {
    this.metricsPath = metricsPath;
    this.writeQueue = [];
    this.isWriting = false;
  }

  async recordMetric(metric) {
    this.writeQueue.push(metric);

    if (!this.isWriting) {
      await this.flush();
    }
  }

  async flush() {
    if (this.writeQueue.length === 0) return;

    this.isWriting = true;
    const metricsToWrite = [...this.writeQueue];
    this.writeQueue = [];

    try {
      // Read existing metrics
      let metrics = {};
      try {
        const data = await fs.readFile(this.metricsPath, 'utf-8');
        metrics = JSON.parse(data);
      } catch (err) {
        if (err.code !== 'ENOENT') throw err;
      }

      // Merge new metrics
      for (const metric of metricsToWrite) {
        metrics[metric.key] = {
          ...metrics[metric.key],
          ...metric.data,
          lastUpdated: new Date().toISOString()
        };
      }

      // Atomic write: write to temp file then rename
      const tempPath = `${this.metricsPath}.tmp`;
      await fs.writeFile(tempPath, JSON.stringify(metrics, null, 2));
      await fs.rename(tempPath, this.metricsPath);

    } catch (error) {
      console.error('Failed to write metrics:', error);
      // Re-queue failed metrics
      this.writeQueue.unshift(...metricsToWrite);
    } finally {
      this.isWriting = false;

      // Process any metrics that arrived during write
      if (this.writeQueue.length > 0) {
        setTimeout(() => this.flush(), 100);
      }
    }
  }
}
Debugging Workflows

Step-by-Step Debugging Process:

  1. Check recent logs for errors
  2. Verify database connectivity
  3. Inspect workflow state in database
  4. Review agent execution traces
  5. Check resource usage (CPU, memory, disk)
# 1. Check recent error logs
tail -n 100 logs/error.log | grep -i "error\|fail\|exception"

# 2. Verify database connection
node -e "require('mongoose').connect(process.env.MONGODB_URI).then(() => console.log('Connected')).catch(e => console.error(e))"

# 3. Inspect stuck workflows
db.workflows.find({ status: 'running', updatedAt: { $lt: new Date(Date.now() - 3600000) } })

# 4. Check agent execution traces
db.agentExecutions.find({ workflowId: '<workflow_id>' }).sort({ startTime: -1 })

# 5. Monitor resource usage
top -p $(pgrep -f "node.*server")
df -h data/metrics/
free -m
Performance Troubleshooting

Common Performance Issues:

  • Slow API responses: Check database query performance, add indexes
  • High CPU usage: Profile Node.js event loop, check for blocking operations
  • Memory growth: Take heap snapshots, look for retained objects
  • Slow workflow execution: Parallelize independent agents, cache results
// Profile API endpoint performance
const express = require('express');
const app = express();

app.use((req, res, next) => {
  const start = process.hrtime.bigint();

  res.on('finish', () => {
    const end = process.hrtime.bigint();
    const durationMs = Number(end - start) / 1e6;

    if (durationMs > 1000) {
      console.warn(`Slow request: ${req.method} ${req.path} took ${durationMs.toFixed(2)}ms`);
    }

    // Record metrics
    metrics.recordHistogram('http_request_duration_ms', durationMs, {
      method: req.method,
      path: req.path,
      status: res.statusCode
    });
  });

  next();
});

// Add database query logging
mongoose.set('debug', (collectionName, methodName, ...args) => {

});

node scripts/stop-workflows.js --graceful

pkill -f "node.*workflow"

# Stop API server
pm2 stop api-server

# Backup current state
cp data/metrics/metrics.json data/metrics/metrics.json.backup
mongodump --uri="$MONGODB_URI" --out=backup/

# Find previous stable version
git log --oneline -10

# Checkout previous version
git checkout <commit-hash>

# Rebuild and restart
npm install
npm run build
